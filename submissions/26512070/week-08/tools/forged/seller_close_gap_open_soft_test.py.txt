def test():
    base = {"kind": "goods", "max_turns": 12, "name": "x"}
    v = dict(base, list_price=100000, cost=60000, floor=70000, ask=95000, offer=80000, turn=2, known=True)
    r = decide(v)
    assert r == {"action": "counter", "price": 85250}
    v = dict(base, list_price=100000, cost=60000, floor=70000, ask=95000, offer=65000, turn=2, known=True)
    r = decide(v)
    assert r["action"] == "counter" and r["price"] >= 70000
    v = dict(base, list_price=100000, cost=60000, floor=70000, ask=94000, offer=84000, turn=2, known=True)
    assert decide(v) == {"action": "accept"}
    v = dict(base, list_price=20000, cost=12000, floor=15000, ask=None, offer=None, turn=1, known=False)
    assert decide(v) == {"action": "counter", "price": 18000}
    v = dict(base, list_price=20000, cost=12000, floor=19000, ask=None, offer=None, turn=1, known=False)
    assert decide(v) == {"action": "counter", "price": 19000}
    v = dict(base, kind="food", list_price=20000, cost=12000, floor=15000, ask=None, offer=None, turn=1, known=False)
    assert decide(v) is None
